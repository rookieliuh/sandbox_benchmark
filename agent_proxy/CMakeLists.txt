cmake_minimum_required(VERSION 3.20)
project(agent_proxy VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Agent Proxy v1 targets Linux. See README.md for server build instructions.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Protobuf REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBCURL REQUIRED IMPORTED_TARGET libcurl)
find_path(GFLAGS_INCLUDE_DIR gflags/gflags.h REQUIRED)
find_library(GFLAGS_LIBRARY NAMES gflags REQUIRED)
find_library(LEVELDB_LIBRARY NAMES leveldb REQUIRED)
find_library(PROTOC_LIBRARY NAMES protoc REQUIRED)
set(BRPC_ROOT "" CACHE PATH "bRPC install prefix (1.12.1 recommended)")
find_path(BRPC_INCLUDE_DIR brpc/server.h HINTS ${BRPC_ROOT}/include REQUIRED)
find_library(BRPC_LIBRARY NAMES brpc HINTS ${BRPC_ROOT}/lib ${BRPC_ROOT}/lib64 REQUIRED)
add_library(brpc_dependency INTERFACE)
target_include_directories(brpc_dependency INTERFACE ${BRPC_INCLUDE_DIR} ${GFLAGS_INCLUDE_DIR})
target_link_libraries(brpc_dependency INTERFACE ${BRPC_LIBRARY} protobuf::libprotobuf
  ${GFLAGS_LIBRARY} ${LEVELDB_LIBRARY} ${PROTOC_LIBRARY} OpenSSL::SSL OpenSSL::Crypto
  ZLIB::ZLIB Threads::Threads ${CMAKE_DL_LIBS} rt)

protobuf_generate_cpp(PROTO_SOURCES PROTO_HEADERS proto/agent_proxy.proto)
add_library(agent_proxy_proto STATIC ${PROTO_SOURCES} ${PROTO_HEADERS})
target_include_directories(agent_proxy_proto PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(agent_proxy_proto PUBLIC protobuf::libprotobuf)
add_library(agent_proxy_core STATIC
  src/core/model.cc src/tools/arguments.cc src/config/config.cc src/transport/http.cc
  src/adapters/e2b/connect_codec.cc src/adapters/e2b/e2b_adapter.cc)
target_include_directories(agent_proxy_core PUBLIC src)
target_link_libraries(agent_proxy_core PUBLIC nlohmann_json::nlohmann_json
  PkgConfig::LIBCURL OpenSSL::Crypto Threads::Threads)
target_compile_options(agent_proxy_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(agent-proxy src/main.cc src/service/service.cc)
target_link_libraries(agent-proxy PRIVATE agent_proxy_core agent_proxy_proto brpc_dependency)
target_compile_options(agent-proxy PRIVATE -Wall -Wextra -Wpedantic)
add_executable(agent-proxy-client client/main.cc)
target_link_libraries(agent-proxy-client PRIVATE agent_proxy_proto brpc_dependency)
target_compile_options(agent-proxy-client PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_executable(core-tests tests/unit/core_test.cc)
  target_link_libraries(core-tests PRIVATE agent_proxy_core)
  add_test(NAME core COMMAND core-tests)
  add_test(NAME mock-protocol COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s ${CMAKE_CURRENT_SOURCE_DIR}/tests/mock_backend -p "test_*.py" -v)
  add_test(NAME e2e COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/e2e.py
    --proxy $<TARGET_FILE:agent-proxy> --client $<TARGET_FILE:agent-proxy-client>
    --report ${CMAKE_CURRENT_BINARY_DIR}/test-results/e2e.json)
  set_tests_properties(e2e PROPERTIES TIMEOUT 120)
endif()
